2. Kịch bản SQL Migration (Database Schema & RLS)

            
          

          

            Mở mục SQL Editor trên Supabase Dashboard và dán toàn bộ đoạn mã bên dưới để khởi tạo 2 bảng posts, comments cùng Storage bucket garden-files:
          


          
-- 1. Bảng Bài viết & Ý tưởng (posts)
CREATE TABLE IF NOT EXISTS posts (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  title TEXT NOT NULL,
  category TEXT NOT NULL CHECK (category IN ('research', 'ideas', 'stories')),
  growth_stage TEXT DEFAULT 'seedling' CHECK (growth_stage IN ('seedling', 'budding', 'evergreen')),
  tags TEXT[] DEFAULT '{}',
  summary TEXT,
  content TEXT NOT NULL,
  bibtex TEXT,
  file_url TEXT,
  claps_count INT DEFAULT 0,
  created_at TIMESTAMPTZ DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- 2. Bảng Bình luận (comments)
CREATE TABLE IF NOT EXISTS comments (
  id UUID DEFAULT gen_random_uuid() PRIMARY KEY,
  post_id UUID REFERENCES posts(id) ON DELETE CASCADE,
  author_name TEXT NOT NULL,
  content TEXT NOT NULL,
  created_at TIMESTAMPTZ DEFAULT timezone('utc'::text, now()) NOT NULL
);

-- 3. Tạo Storage Bucket lưu tệp
INSERT INTO storage.buckets (id, name, public) 
VALUES ('garden-files', 'garden-files', true)
ON CONFLICT (id) DO NOTHING;

-- 4. Kích hoạt Row Level Security (RLS)
ALTER TABLE posts ENABLE ROW LEVEL SECURITY;
ALTER TABLE comments ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Public Read Posts" ON posts FOR SELECT USING (true);
CREATE POLICY "Public Read Comments" ON comments FOR SELECT USING (true);
CREATE POLICY "Public Insert Comments" ON comments FOR INSERT WITH CHECK (true);
CREATE POLICY "Public Update Claps" ON posts FOR UPDATE USING (true) WITH CHECK (true);
CREATE POLICY "Public Insert Posts" ON posts FOR INSERT WITH CHECK (true);

-- Storage Policies
CREATE POLICY "Public Access Bucket" ON storage.objects FOR SELECT USING (bucket_id = 'garden-files');
CREATE POLICY "Public Upload Bucket" ON storage.objects FOR INSERT WITH CHECK (bucket_id = 'garden-files');
